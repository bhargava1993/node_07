2.$group, $sum, $avg,  $max, $min

1. calculate total order amount

db.orders.aggregate([

    { $group: {
        _id:null,
        totalAmount: {
            $sum: "$amount"
        }
    }}

])

2. calculate total amount for each customer

db.orders.aggregate([
    {
        $group:{
            _id:"$customer",
             totalAmount: {
            $sum: "$amount"
          }
        }
    }
])


3.avg

db.orders.aggregate([
    {
        $group:{
            _id:"$customer",
             averageAmount: {
               $avg: "$amount"
          }
        }
    }
])

4.$max, $min

db.orders.aggregate([
    {
        $group:{
            _id:"null",
             highestAmount: {
               $max: "$amount"
          }
        }
    }
])

5. count orders per customer using $sum:1

db.orders.aggregate([
    {
        $group:{
            _id:"$customer",
             totalOrders: {
               $sum: 1
          }
        }
    }
])

3.$sort, $limit, $skip

$sort :   
-1: descending order,
 1: ascending order

1. sort orders by amount, Highest first

db.orders.find({},{_id:0, customer:1, amount:1}).sort({amount:1})
db.orders.find({},{_id:0, customer:1, amount:1}).sort({amount:-1})

db.orders.aggregate([
    {
        $sort:{
            amount:-1
        }
    },
    {
        $project: {
            _id:0, customer:1, amount:1
        }
    }
])

db.orders.aggregate([
    {
        $sort:{
            amount:1
        }
    },
    {
        $project: {
            _id:0, customer:1, amount:1
        }
    }
])

2. find the top 3 Highest value orders

db.orders.aggregate([
    {
        $sort:{
            amount: -1
        }
    },
    {$limit: 1},
    {
        $project: {
            _id:0, customer:1, amount:1
        }
    }
])

db.orders.aggregate([
